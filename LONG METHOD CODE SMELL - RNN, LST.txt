class _RecurrentNet(nn.Module):
    def __init__(self, n_features, cell, emb, hidden, dropout):
        super().__init__()
        # embedding: metric j with value v becomes the vector v * W_j + b_j
        self.w = nn.Parameter(torch.randn(n_features, emb) * 0.1)
        self.b = nn.Parameter(torch.zeros(n_features, emb))
        self.bidirectional = cell in ("bilstm", "bigru")
        if cell == "rnn":
            self.rnn = nn.RNN(emb, hidden, batch_first=True, nonlinearity="tanh")
        elif cell in ("gru", "bigru"):
            self.rnn = nn.GRU(emb, hidden, batch_first=True,
                              bidirectional=self.bidirectional)
        else:
            self.rnn = nn.LSTM(emb, hidden, batch_first=True,
                               bidirectional=self.bidirectional)
        self.drop = nn.Dropout(dropout)
        self.head = nn.Linear(hidden * (2 if self.bidirectional else 1), 1)

    def forward(self, x):                                   # x: (rows, features)
        seq = x.unsqueeze(-1) * self.w + self.b             # (rows, features, emb)
        _, state = self.rnn(seq)
        h = state[0] if isinstance(state, tuple) else state  # LSTM returns (h, c)
        h = torch.cat([h[-2], h[-1]], dim=1) if self.bidirectional else h[-1]
        return self.head(self.drop(h)).squeeze(1)          # one logit per row


class RecurrentClassifier(ClassifierMixin, BaseEstimator):
    """cell = 'rnn', 'lstm', 'bilstm', 'gru' or 'bigru'."""

    def __init__(self, cell="lstm", emb=16, hidden=32, dropout=0.3, lr=1e-3,
                 batch_size=32, max_epochs=150, patience=15, val_share=0.15,
                 random_state=SEED):
        self.cell = cell
        self.emb = emb
        self.hidden = hidden
        self.dropout = dropout
        self.lr = lr
        self.batch_size = batch_size
        self.max_epochs = max_epochs
        self.patience = patience
        self.val_share = val_share
        self.random_state = random_state

    def fit(self, X, y):
        torch.manual_seed(self.random_state)
        rng = np.random.default_rng(self.random_state)
        X = np.asarray(X, dtype=np.float32)
        y = np.asarray(y)
        self.classes_ = np.unique(y)
        y01 = (y == self.classes_[1]).astype(np.float32)

        # early stopping watches a validation slice of the training rows
        Xa, Xv, ya, yv = train_test_split(X, y01, test_size=self.val_share,
                                          stratify=y01, random_state=self.random_state)
        Xa, Xv, ya, yv = map(torch.tensor, (Xa, Xv, ya, yv))

        self.net_ = _RecurrentNet(X.shape[1], self.cell, self.emb, self.hidden,
                                  self.dropout)
        opt = torch.optim.Adam(self.net_.parameters(), lr=self.lr, weight_decay=1e-4)
        loss_fn = nn.BCEWithLogitsLoss()
        best, best_state, waited = np.inf, None, 0
        for epoch in range(self.max_epochs):
            self.net_.train()
            order = rng.permutation(len(Xa))
            for s in range(0, len(order), self.batch_size):
                idx = order[s:s + self.batch_size]
                opt.zero_grad()
                loss_fn(self.net_(Xa[idx]), ya[idx]).backward()   # forward + backward
                opt.step()                                         # update weights
            self.net_.eval()
            with torch.no_grad():
                v = loss_fn(self.net_(Xv), yv).item()
            if v < best - 1e-4:
                best, waited = v, 0
                best_state = {k: t.clone() for k, t in self.net_.state_dict().items()}
            else:
                waited += 1
                if waited >= self.patience:
                    break
        self.net_.load_state_dict(best_state)
        self.epochs_used_ = epoch + 1
        return self

    def predict_proba(self, X):
        self.net_.eval()
        with torch.no_grad():
            p = torch.sigmoid(self.net_(torch.tensor(np.asarray(X, dtype=np.float32)))).numpy()
        return np.column_stack([1 - p, p])

    def predict(self, X):
        return self.classes_[(self.predict_proba(X)[:, 1] >= 0.5).astype(int)]

def information_gain(X, y):
    """Mutual information between each metric and the label."""
    return mutual_info_classif(X, y, random_state=SEED)


def fisher_score(X, y):
    """Between-class spread of the means divided by the within-class spread."""
    X, y = np.asarray(X, dtype=float), np.asarray(y)
    overall = X.mean(axis=0)
    between = np.zeros(X.shape[1])
    within = np.zeros(X.shape[1])
    for label in np.unique(y):
        rows = X[y == label]
        between += len(rows) * (rows.mean(axis=0) - overall) ** 2
        within += len(rows) * rows.var(axis=0)
    return between / (within + 1e-12)

def build_pipeline(cell, selector_name):
    steps = [("log",   FunctionTransformer(np.log1p)),     # tame very skewed metrics
             ("scale", MinMaxScaler())]                     # every metric to 0-1
    score_func = SELECTORS[selector_name]
    if score_func is not None:
        steps.append(("select", SelectKBest(score_func, k=K_FEATURES)))   # keep best K
    steps += [("smote", SMOTE(random_state=SEED)),          # balance training rows only
              ("model", RecurrentClassifier(cell=cell))]
    return Pipeline(steps)

def draw_metric_bars(metric):
    fig, ax = plt.subplots(figsize=(12, 5))
    pos = np.arange(len(model_order))
    w = 0.8 / len(sel_order)
    floor = max(0, cv[metric].min() - 0.15)
    for i, sel in enumerate(sel_order):
        vals = cv.loc[sel, metric].loc[model_order].values
        errs = cv.loc[sel, metric + "_sd"].loc[model_order].values
        bars = ax.bar(pos + (i - (len(sel_order) - 1) / 2) * w, vals, w * 0.95, yerr=errs,
                      capsize=3, color=COLOURS[sel], label=sel)
        for bar, v in zip(bars, vals):
            ax.text(bar.get_x() + bar.get_width() / 2, floor + 0.01, f"{v:.3f}",
                    ha="center", va="bottom", fontsize=8, rotation=90, color="white")
    ax.set_xticks(pos)
    ax.set_xticklabels(model_order, fontsize=11)
    ax.set_ylim(floor, 1.0)
    ax.set_ylabel(f"{LABELS[metric]} (cross-validation mean +/- sd)")
    ax.set_title(f"{LABELS[metric]} by model and feature selection method - Large Class",

ALPHA = 0.05

all_models = list(folds_df["model"].unique())


def scores(model, selector):
    """MCC of one model + selector, ordered by fold (1..10)."""
    rows = folds_df[(folds_df.model == model) & (folds_df.selector == selector)]
    return rows.sort_values("fold")[METRIC].values


def wilcoxon_p(a, b):
    """Two-sided Wilcoxon signed-rank p-value (1.0 if the folds are identical)."""
    if np.allclose(a - b, 0):
        return 1.0
    return wilcoxon(a, b).pvalue


def add_holm(table):
    """Holm-corrected p-values for one family of comparisons."""
    reject, p_holm, _, _ = multipletests(table["p"], alpha=ALPHA, method="holm")
    table["p_holm"] = p_holm
    table["significant (Holm)"] = np.where(reject, "yes", "no")
    return table


# ---- 1. Friedman: are the models different, within each feature set? ----
friedman = []
for sel in sel_order:
    stat, p = friedmanchisquare(*[scores(m, sel) for m in all_models])
    friedman.append({"feature set": sel, "chi2": stat, "p": p})
friedman = add_holm(pd.DataFrame(friedman))
print("\n1. FRIEDMAN TEST - all models compared within each feature set")
print(friedman.round(4).to_string(index=False))

# ---- 2. Wilcoxon: each feature-selection method vs all features ----
fs = []
for m in all_models:
    base = scores(m, "All features")
    for sel in sel_order[1:]:
        s = scores(m, sel)
        fs.append({"model": m, "selector": sel,
                   "mean diff": (s - base).mean(),
                   "better folds": int((s > base).sum()),
                   "worse folds": int((s < base).sum()),
                   "p": wilcoxon_p(s, base)})
fs = add_holm(pd.DataFrame(fs))
print("\n2. WILCOXON - feature selection vs all features")
print(fs.round(4).to_string(index=False))

# ---- 3. Wilcoxon: bidirectional vs unidirectional ----
bi = []
for b, u in [("BiLSTM", "LSTM"), ("BiGRU", "GRU")]:
    for sel in sel_order:
        sb, su = scores(b, sel), scores(u, sel)
        bi.append({"comparison": f"{b} vs {u}", "feature set": sel,
                   "mean diff": (sb - su).mean(), "p": wilcoxon_p(sb, su)})
bi = add_holm(pd.DataFrame(bi))
print("\n3. WILCOXON - bidirectional vs unidirectional")
print(bi.round(4).to_string(index=False))

# ---- 4. Wilcoxon: best recurrent model vs TabPFN (same feature set) ----
vs_tab = None
if "TabPFN" in all_models:
    rec = [m for m in all_models if m != "TabPFN"]
    vs_tab = []
    for sel in sel_order:
        best_rec = max(rec, key=lambda m: scores(m, sel).mean())
        sr, st = scores(best_rec, sel), scores("TabPFN", sel)
        vs_tab.append({"feature set": sel, "best recurrent": best_rec,
                       "mean diff (rec - TabPFN)": (sr - st).mean(),
                       "p": wilcoxon_p(sr, st)})
    vs_tab = add_holm(pd.DataFrame(vs_tab))
    print("\n4. WILCOXON - best recurrent model vs TabPFN")
    print(vs_tab.round(4).to_string(index=False))
